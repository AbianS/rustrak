<Section n="03" title="Guidelines">

<DoDont>
  <Do example={<div className="flex items-center gap-3"><Checkbox aria-label="Select row" /><Text variant="control">Select this issue</Text></div>}>
    A name for it, always. It carries no visible text of its own.
  </Do>
  <Dont example={<Checkbox aria-label="" />}>
    An unnamed box. A screen reader announces “checkbox” and nothing else.
  </Dont>
</DoDont>

<DoDont>
  <Do example={<div className="flex items-center gap-3"><Checkbox aria-label="Select all" indeterminate /><Text variant="column" tone="meta">Select all</Text></div>}>
    Indeterminate only on a select-all header.
  </Do>
  <Dont example={<div className="flex items-center gap-3"><Checkbox aria-label="Maybe" indeterminate /><Text variant="control">Email me about this</Text></div>}>
    Indeterminate as a user-facing option. There is no “maybe”.
  </Dont>
</DoDont>

</Section>
